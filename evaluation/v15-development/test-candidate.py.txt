                    body="Checklist updated.",
                    created_at="2026-10-23T10:00:00Z",
                    updated_at="2026-10-23T10:00:00Z",
                ),
            ],
        )
    )


def candidates():
    return [
        Candidate(
            request_id="r",
            request_quote=quote,
            response_id=None,
            response_quote=None,
            category="change",
            interpretation="follow_up",
            rationale="Untrusted extraction rationale",
            uncertainty=["Inspect context"],
        )
        for quote in [
            "Please update the checklist.",
            "I already refreshed the reference image.",
        ]
    ]


def output():
    return ModelOutput.model_validate(
        {
            "cards": [
                {
                    "request_passage": f"r/p{i}",
                    "category": "change",
                    "rationale": "Proposed ask.",
                    "uncertainty": [],
                }
                for i in range(2)
            ]
        }
    )


def configured(monkeypatch, tmp_path):
    model = LocalModel(Settings(root=tmp_path))
    monkeypatch.setattr(model, "_credential", lambda: "synthetic-test-key")
    monkeypatch.setattr(
        model,
        "_request",
        lambda *_args, **_kwargs: {
            "data": [{"id": model.settings.model_name, "meta": {"n_ctx": 16384}}],
        },
    )
    return model


def test_verifier_context_keeps_complete_reviewer_comments_as_untrusted_data():
    messages = verification_messages_for(example(), candidates())
    evidence = json.loads(messages[1]["content"])
    assert messages[0]["role"] == "system" and messages[1]["role"] == "user"
    assert evidence["pr_title"] == example().title
    assert [source["id"] for source in evidence["reviewer_comments"]] == ["r"]
    assert evidence["request_candidates"] == [
        {
            "request_index": index,
            "source_comment_id": candidate.request_id,
            "quote": candidate.request_quote,
        }
        for index, candidate in enumerate(candidates())
    ]
    assert "author_updates" not in evidence
    assert "Untrusted extraction rationale" not in messages[1]["content"]


def test_schema_limits_verifier_to_unique_known_integer_indices():
    schema = constrained_schema(VerifiedRequests, example(), candidates())
    field = schema["properties"]["keep_request_indices"]
    assert field["items"] == {"type": "integer", "enum": [0, 1]}
    assert field["maxItems"] == 2 and field["uniqueItems"] is True
    assert schema["additionalProperties"] is False
    for invalid in [[True], ["0"], [None]]:
        with pytest.raises(ValidationError):
            VerifiedRequests.model_validate({"keep_request_indices": invalid})
    with pytest.raises(ValidationError):
        VerifiedRequests.model_validate(
            {"keep_request_indices": [0], "quote": "rewrite"}
        )


@pytest.mark.parametrize("indices", [[-1], [2], [0, 0]])
def test_local_binding_rejects_runtime_schema_bypass(indices):
    with pytest.raises(ModelFailure, match="invalid or repeated request indices"):
        bind_verified_requests(
            VerifiedRequests(keep_request_indices=indices), candidates()
        )


def test_verifier_can_only_select_original_objects_without_reordering_or_mutation():
    proposed = candidates()
    before = [candidate.model_dump() for candidate in proposed]
    selected = bind_verified_requests(
        VerifiedRequests(keep_request_indices=[1, 0]), proposed
    )
    assert selected[0] is proposed[0] and selected[1] is proposed[1]
    assert [candidate.model_dump() for candidate in proposed] == before
    assert (
        bind_verified_requests(VerifiedRequests(keep_request_indices=[]), proposed)
        == []
    )


def test_nonempty_extraction_is_verified_before_only_kept_requests_are_matched(
    monkeypatch, tmp_path
):
    model = configured(monkeypatch, tmp_path)
    calls = []

    def generate(
        _client,
        _messages,
        schema,
        _budget,
        _pr,
        requests=None,
        focus_comment_id=None,
        deadline=None,
    ):
        calls.append((schema, requests, deadline))
        if schema is ModelOutput:
            return output()
        if schema is VerifiedRequests:
            assert len(requests) == 2
            return VerifiedRequests(keep_request_indices=[0])
        assert len(requests) == 1
        return ResponseLinks.model_validate(
            {
                "links": {
                    "request_0": {
                        "response_passage": "a/p0",
                        "rationale": "Author reports updating the checklist.",
                        "uncertainty": ["Not verified"],
                    }
                }
            }
        )

    monkeypatch.setattr(model, "_generate", generate)
    result = model.analyze(example())
    assert [schema for schema, _, _ in calls] == [
        ModelOutput,
        VerifiedRequests,
        ResponseLinks,
    ]
    assert len({deadline for _, _, deadline in calls}) == 1
    assert len(result) == 1 and result[0].request_quote == candidates()[0].request_quote
    assert result[0].response_quote == "Checklist updated."

def test_verifier_consumes_same_remaining_pr_deadline(monkeypatch, tmp_path):
    model = configured(monkeypatch, tmp_path)
    clock = [10.0]
    monkeypatch.setattr("reviewradar.model.time.monotonic", lambda: clock[0])
    calls = []

    def generate(
        _client,
        _messages,
        schema,
        _budget,
        _pr,
        requests=None,
        focus_comment_id=None,
        deadline=None,
    ):
        calls.append(schema)
        if schema is ModelOutput:
            return output()
        assert schema is VerifiedRequests
        clock[0] = deadline + 1
        return VerifiedRequests(keep_request_indices=[])

    monkeypatch.setattr(model, "_generate", generate)
    with pytest.raises(ModelFailure, match="whole-PR analysis time budget"):
        model.analyze(example())
    assert calls == [ModelOutput, VerifiedRequests]


def test_verification_prompt_still_requires_actual_token_budget_before_generation(
    monkeypatch, tmp_path
):
    model = configured(monkeypatch, tmp_path)
    calls = []

    def post(_client, path, _payload, *, deadline=None):
        calls.append(path)
        return (
            {"prompt": "Full untruncated reviewer context"}
            if path == "/apply-template"
            else {"tokens": [1] * 101}
        )

    monkeypatch.setattr(model, "_post", post)
    with httpx.Client() as client, pytest.raises(Oversize, match="not truncated"):
        model._generate(
            client,
            verification_messages_for(example(), candidates()),
            VerifiedRequests,
            100,
            example(),
            candidates(),
        )
    assert calls == ["/apply-template", "/tokenize"]
